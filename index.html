// ============================================================ // UpGro Media — Google Sheets API Bridge v3.0 FINAL // FIXED: Smart date detection even without "Date Applied" column // Works with ANY column structure // ============================================================ var START_DATE = new Date('2026-04-01'); // ← Change this if needed function doGet(e) { try { var params = e.parameter || {}; var action = params.action || 'getAllCandidates'; var startDate = params.startDate ? new Date(params.startDate) : START_DATE; var callback = params.callback; var result; if (action === 'getAllCandidates') result = getCandidates(startDate); else if (action === 'ping') result = { status: 'ok', time: new Date().toISOString() }; else if (action === 'getSheetNames') result = getSheetNames(); else result = { error: 'Unknown action: ' + action }; var json = JSON.stringify(result); if (callback) { return ContentService.createTextOutput(callback + '(' + json + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(json) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput( JSON.stringify({ error: err.message, stack: err.stack }) ).setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { try { var data = JSON.parse(e.postData.contents); var result; if (data.action === 'updateField') result = updateField(data.rowId, data.field, data.value); else if (data.action === 'updateStatus') result = updateStatus(data.rowId, data.status); else if (data.action === 'updateSalary') { result = updateMultipleFields(data.rowId, { 'Final Salary': data.salary, 'Final Designation': data.role, 'Joining Date': data.joiningDate }); } else result = { success: false, error: 'Unknown action' }; return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput( JSON.stringify({ success: false, error: err.message }) ).setMimeType(ContentService.MimeType.JSON); } } // ── MAIN: GET ALL CANDIDATES with smart date filter ────────── function getCandidates(startDate) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheets = ss.getSheets(); var all = []; var skipped = 0; var errors = []; // Skip these sheet names (config/helper sheets) var SKIP_SHEETS = ['Config', 'Settings', 'Template', 'Instructions', 'Sheet1', 'Sheet2', 'Sheet3']; sheets.forEach(function(sheet) { try { var name = sheet.getName(); if (SKIP_SHEETS.indexOf(name) > -1) return; var lastRow = sheet.getLastRow(); var lastCol = sheet.getLastColumn(); if (lastRow < 2 || lastCol < 1) return; var data = sheet.getRange(1, 1, lastRow, lastCol).getValues(); var headers = data[0].map(function(h) { return String(h).trim(); }); // ── Find date column (checks many possible names) ── var DATE_COLUMNS = [ 'Date Applied', 'Applied Date', 'Date', 'Timestamp', 'Submission Date', 'Apply Date', 'Application Date', 'Submit Date', 'Entry Date', 'Date of Application', 'Applied On', 'Received Date', 'Form Date' ]; var dateColIdx = -1; for (var d = 0; d < DATE_COLUMNS.length; d++) { var idx = findColIndex(headers, DATE_COLUMNS[d]); if (idx > -1) { dateColIdx = idx; break; } } // ── Find name column ── var NAME_COLUMNS = ['Full Name', 'Name', 'Candidate Name', 'Applicant Name', 'Student Name']; var nameColIdx = findColAny(headers, NAME_COLUMNS); if (nameColIdx === -1) nameColIdx = 0; // fallback to first column for (var i = 1; i < data.length; i++) { var row = data[i]; // Skip empty rows var nameVal = row[nameColIdx]; if (!nameVal || String(nameVal).trim() === '') { skipped++; continue; } // ── DATE FILTER LOGIC ────────────────────────────────── var passedDateFilter = false; if (dateColIdx > -1) { // Has a date column — use it var rawDate = row[dateColIdx]; if (rawDate === '' || rawDate === null || rawDate === undefined) { // Date column exists but this row is EMPTY → INCLUDE IT // (data entry mein date nahi dali — include karo, reject mat karo) passedDateFilter = true; } else { var parsed = parseAnyDate(rawDate); if (parsed && parsed >= startDate) { passedDateFilter = true; } else if (!parsed) { // Can't parse date → include it (better to show than miss) passedDateFilter = true; } } } else { // NO date column at all → include all rows // (Agar date column hi nahi hai to sab dikhao) passedDateFilter = true; } if (!passedDateFilter) { skipped++; continue; } // ── Build row object ── var rowObj = {}; headers.forEach(function(h, j) { if (h) rowObj[h] = row[j] !== undefined ? formatCellValue(row[j]) : ''; }); rowObj['_rowId'] = name + '_' + i; rowObj['_sheetName'] = name; rowObj['_rowIndex'] = i; rowObj['_dateFound'] = dateColIdx > -1 ? 'yes' : 'no_date_column'; all.push(rowObj); } } catch (sheetErr) { errors.push(sheet.getName() + ': ' + sheetErr.message); } }); return { success: true, candidates: all, total: all.length, skipped: skipped, sheetCount: sheets.length, startDate: startDate.toISOString().split('T')[0], fetchedAt: new Date().toISOString(), errors: errors }; } // ── Parse any date format India uses ───────────────────────── function parseAnyDate(val) { if (!val) return null; // Already a Date object (Google Sheets returns Date objects) if (val instanceof Date && !isNaN(val.getTime())) return val; var s = String(val).trim(); if (!s) return null; // DD/MM/YYYY or DD-MM-YYYY (Indian format) var ddmmyyyy = s.match(/^(\d{1,2})[\/\-\.](\d{1,2})[\/\-\.](\d{4})$/); if (ddmmyyyy) { var d = new Date(parseInt(ddmmyyyy[3]), parseInt(ddmmyyyy[2])-1, parseInt(ddmmyyyy[1])); if (!isNaN(d)) return d; } // YYYY-MM-DD (ISO format) var iso = s.match(/^(\d{4})[\/\-\.](\d{1,2})[\/\-\.](\d{1,2})$/); if (iso) { var d2 = new Date(parseInt(iso[1]), parseInt(iso[2])-1, parseInt(iso[3])); if (!isNaN(d2)) return d2; } // MM/DD/YYYY (US format) var us = s.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/); if (us) { var d3 = new Date(parseInt(us[3]), parseInt(us[1])-1, parseInt(us[2])); if (!isNaN(d3)) return d3; } // Try native parse as fallback var native = new Date(s); if (!isNaN(native.getTime())) return native; return null; // Can't parse } // ── Format cell value for JSON ──────────────────────────────── function formatCellValue(val) { if (val instanceof Date) { // Return as DD/MM/YYYY string var dd = String(val.getDate()).padStart(2, '0'); var mm = String(val.getMonth()+1).padStart(2, '0'); var yyyy = val.getFullYear(); return dd + '/' + mm + '/' + yyyy; } return val; } // ── Column helpers ──────────────────────────────────────────── function findColIndex(headers, name) { var lower = name.toLowerCase().trim(); for (var i = 0; i < headers.length; i++) { if (headers[i].toLowerCase().trim() === lower) return i; } return -1; } function findColAny(headers, names) { for (var n = 0; n < names.length; n++) { var idx = findColIndex(headers, names[n]); if (idx > -1) return idx; } return -1; } // ── Update single field ─────────────────────────────────────── function updateField(rowId, field, value) { try { var parts = String(rowId).split('_'); var rowIndex = parseInt(parts[parts.length-1]); var sheetName = parts.slice(0, parts.length-1).join('_'); var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(sheetName); if (!sheet) return { success: false, error: 'Sheet not found: ' + sheetName }; var headers = sheet.getRange(1,1,1,sheet.getLastColumn()).getValues()[0]; var colIdx = findColIndex(headers, field); if (colIdx === -1) { // Column doesn't exist — create it colIdx = sheet.getLastColumn(); sheet.getRange(1, colIdx+1).setValue(field); } sheet.getRange(rowIndex+1, colIdx+1).setValue(value); return { success: true }; } catch (err) { return { success: false, error: err.message }; } } function updateMultipleFields(rowId, fields) { var keys = Object.keys(fields); var results = []; for (var i = 0; i < keys.length; i++) { results.push(updateField(rowId, keys[i], fields[keys[i]])); } return { success: true, updates: results }; } function updateStatus(rowId, status) { return updateField(rowId, 'Status', status); } function getSheetNames() { var ss = SpreadsheetApp.getActiveSpreadsheet(); return { success: true, sheets: ss.getSheets().map(function(s) { return { name: s.getName(), rows: s.getLastRow(), cols: s.getLastColumn() }; }) }; } // ── Run this manually in Apps Script editor to test ────────── function testRun() { var result = getCandidates(new Date('2026-04-01')); Logger.log('Total candidates: ' + result.total); Logger.log('Skipped (before date): ' + result.skipped); Logger.log('Sheets scanned: ' + result.sheetCount); if (result.errors.length) Logger.log('Errors: ' + result.errors.join('; ')); if (result.candidates.length > 0) { Logger.log('First candidate: ' + JSON.stringify(result.candidates[0])); } }
